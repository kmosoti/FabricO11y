"""The anatomy of one Batch from the node's timestamped stdout (spindle_bench.py
writes <work>/node.out): the period between consecutive `batch=` lines, the
sender's own send-to-answer time (`elapsed_us`), and the two halves of the
period: batch line to its ack line (delivery) and ack line to the next batch
line (collection, encoding and the Spool append)."""
import re, statistics, sys
lines = [l.rstrip("\n") for l in open(sys.argv[1])]
bt = [(float(l.split()[0]), l) for l in lines if " batch=" in l]
at = [(float(l.split()[0]), l) for l in lines if "elapsed_us=" in l]
el = [int(re.search(r"elapsed_us=(\d+)", l).group(1)) for _, l in at]
per = [b - a for (a, _), (b, _) in zip(bt, bt[1:])]
print(f"batches {len(bt)}; period per Batch median {statistics.median(per) * 1000:.1f} ms")
print(f"send-to-answer median {statistics.median(el) / 1000:.1f} ms, p10 {sorted(el)[len(el) // 10] / 1000:.1f}, p90 {sorted(el)[int(len(el) * .9)] / 1000:.1f}")
ca, an, ai = [], [], 0
for i, (t, _) in enumerate(bt):
    while ai < len(at) and at[ai][0] < t:
        ai += 1
    if ai < len(at):
        ca.append(at[ai][0] - t)
        if i + 1 < len(bt):
            an.append(bt[i + 1][0] - at[ai][0])
print(f"batch line to ack line median {statistics.median(ca) * 1000:.1f} ms; ack line to next batch line median {statistics.median(an) * 1000:.1f} ms")
