import importlib.util
import json
import os
from pathlib import Path
from tempfile import TemporaryDirectory
from unittest.mock import patch
import sys

root=Path('/home/kmosoti/projects/fabric_o11y/tools/seal-probe')
sys.path.insert(0,str(root))
spec=importlib.util.spec_from_file_location('cost_bench',root/'bench.py')
bench=importlib.util.module_from_spec(spec)
spec.loader.exec_module(bench)
checks=0

with TemporaryDirectory(dir='/tmp/fabric-s0-e2-verify') as name:
    temp=Path(name)
    actual={}
    for variant in ('one','two'):
        path=temp/f'{variant}.seal'
        calls=[]
        original_write=os.write
        original_fsync=os.fsync
        def traced_write(fd,data):
            calls.append(('write',len(data)))
            return original_write(fd,data)
        def traced_fsync(fd):
            kind='file' if os.path.sameopenfile(fd,file_fd) else 'dir'
            calls.append(('fsync',kind))
            return original_fsync(fd)
        original_open=os.open
        file_fd=None
        def traced_open(p, flags, *args):
            global file_fd
            fd=original_open(p,flags,*args)
            if Path(p)==path:
                file_fd=fd
            return fd
        with patch.object(bench.os,'open',traced_open), patch.object(bench.os,'write',traced_write), patch.object(bench.os,'fsync',traced_fsync):
            result=bench.write_trial(path,512,1,variant,1)
        assert bench.verify_trial(path,512,1,1)['sha256']==bench.sha256(path)
        assert result['successful_fsyncs']==(1 if variant=='one' else 2)
        actual[variant]=[c for c in calls if c!=('fsync','dir')]
        checks+=1
    assert actual['one']==[('write',1024),('write',512),('fsync','file')],actual
    assert actual['two']==[('write',1024),('fsync','file'),('write',512),('fsync','file')],actual
    assert (temp/'one.seal').read_bytes()==(temp/'two.seal').read_bytes()
    checks+=1

    # Failed fsync gives no successful result and no following operation.
    path=temp/'fail.seal'
    calls=[]
    original_fsync=os.fsync
    original_open=os.open
    file_fd=None
    def traced_open(p, flags, *args):
        global file_fd
        fd=original_open(p,flags,*args)
        if Path(p)==path:
            file_fd=fd
        return fd
    def failing_fsync(fd):
        if fd==file_fd:
            calls.append('file_fsync_failed')
            raise OSError('injected EIO')
        return original_fsync(fd)
    with patch.object(bench.os,'open',traced_open), patch.object(bench.os,'fsync',failing_fsync):
        try:
            bench.write_trial(path,512,1,'two',1)
        except OSError as error:
            assert 'injected EIO' in str(error)
        else:
            raise AssertionError('failed sync returned a successful trial')
    assert calls==['file_fsync_failed']
    assert path.stat().st_size==1024 # two-sync did not write seal after failed data sync
    checks+=1

    # A child failure is recorded and propagated.
    out=temp/'child-output'
    out.mkdir()
    cmd=[sys.executable,str(root/'bench.py'),'_verify',str(temp/'missing.seal'),'512','1','1']
    try:
        bench.run_child(cmd,out,'missing-verify')
    except RuntimeError:
        pass
    else:
        raise AssertionError('failed verifier child was accepted')
    record=json.loads((out/'commands.jsonl').read_text())
    assert record['exit_code']==1
    checks+=1

    # Corrupt the writer's reported sync counter after a successful child.
    # The orchestrator must reject it before accepting the first trial.
    original_child=bench.run_child
    def corrupted_child(command,out,stem):
        record=original_child(command,out,stem)
        if stem=='s512-g1-warmup-one-write':
            stream=out/record['stdout']
            data=json.loads(stream.read_text())
            data['successful_fsyncs']=0
            stream.write_text(json.dumps(data))
        return record
    run_out=temp/'counter-corrupt'
    with patch.object(bench,'run_child',corrupted_child):
        try:
            bench.run(run_out,12,smoke=True)
        except ValueError as error:
            assert 'counts differ' in str(error)
        else:
            raise AssertionError('counter corruption accepted')
    assert not (run_out/'smoke.json').exists()
    checks+=1

print(json.dumps({'checks':checks,'write_boundaries':actual,'result':'PASS'}))
