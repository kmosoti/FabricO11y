from pathlib import Path
import importlib.util
import struct
import sys

root = Path('/home/kmosoti/projects/fabric_o11y/tools/seal-probe')
sys.path.insert(0, str(root))
from oracle_support import bodies, group_bytes
spec = importlib.util.spec_from_file_location('candidate', root / 'candidate.py')
model = importlib.util.module_from_spec(spec)
spec.loader.exec_module(model)

cases = 0
for sector in (512, 4096):
    prior_bodies = bodies(1, 11)
    current_bodies = bodies(6, 12)
    prior = group_bytes(0, prior_bodies, sector)
    current = group_bytes(1, current_bodies, sector)
    clean = prior + current
    assert model.recover(clean, sector) == ('ok', (prior_bodies, current_bodies), len(clean))
    cases += 1

    # Change a body extent to a different valid value. A present seal must
    # never be reinterpreted as an absent seal or lose group 0.
    for length in (57, 58, 127, 512, 513, 4096):
        bad = bytearray(clean)
        struct.pack_into('<I', bad, len(prior) + 16, length)
        bad = bytes(bad)
        if bad != clean:
            assert model.recover(bad, sector) == ('error', bad), ('extent', sector, length)
            cases += 1

    # Remove descriptor only and leave complete seal: fail closed.
    bad = bytes(len(prior) + sector) + clean[len(prior) + sector:]
    assert model.recover(bad, sector) == ('error', bad), ('descriptor_missing', sector)
    cases += 1

    # Missing in-flight seal may discard group 1 but must preserve group 0.
    bad = clean[:-sector] + bytes(sector)
    assert model.recover(bad, sector) == ('ok', (prior_bodies,), len(prior)), ('seal_missing', sector)
    cases += 1

    # Corrupt a previously ACKed seal while a later clean group remains.
    bad = bytearray(clean)
    bad[len(prior) - sector + 20] ^= 1
    bad = bytes(bad)
    assert model.recover(bad, sector) == ('error', bad), ('acked_seal', sector)
    cases += 1

    # The same bytes after a failed final sync require different supervision.
    success = model.append_logical(prior, 1, current_bodies, sector, None)
    failed = model.append_logical(prior, 1, current_bodies, sector, 'sync')
    assert success[0] == failed[0] == clean
    scan = model.recover(clean, sector)
    assert model.supervise(False, scan) == 'resume_allowed'
    assert model.supervise(True, scan) == 'rebuild_from_trusted_source'
    assert success[1:4] == (True, False, False)
    assert failed[1:4] == (False, True, True)
    cases += 1
print(f'PASS: {cases} targeted E2 cases')
