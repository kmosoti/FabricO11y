use std::num::NonZeroUsize;
use fabric_o11y::{Event, EventId, TenantId, SourceId, ResourceId, EventTime, ObservedTime, Payload};
use storage_probe::Query;
use storage_probe::coverage::{self, CoverageStatus, CoverageError, Disposition, SealedSnapshot};

fn row(i: usize) -> Event {
    Event {
        id: EventId((i % 2) as u64), tenant: TenantId((i % 3) as u64),
        source: SourceId(i as u64), resource: ResourceId(99),
        event_time: EventTime(if i % 7 == 0 { i64::MIN } else if i % 11 == 0 { i64::MAX } else { i as i64 - 8 }),
        observed_time: ObservedTime(i as i64), attributes: vec![],
        payload: if i % 4 == 0 { Payload::Gauge { name: "g".into(), value: -0.0, unit: "u".into() } }
            else { Payload::Log { body: if i % 2 == 0 { "rare\u{2003}β" } else { "common β" }.into() } },
    }
}
fn exact(e: &Event, q: &Query) -> bool {
    e.event_time.0 >= q.start_ns && e.event_time.0 <= q.end_ns
      && q.tenant.map_or(true, |t| e.tenant.0 == t)
      && q.token.as_ref().map_or(true, |t| match &e.payload {
          Payload::Log { body } => body.split_whitespace().any(|word| word == t),
          _ => false,
      })
}
fn main() {
    let queries = [
        Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: None },
        Query { start_ns: i64::MIN, end_ns: i64::MIN, tenant: None, token: Some("β".into()) },
        Query { start_ns: -2, end_ns: 4, tenant: Some(1), token: Some("common".into()) },
        Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("rare".into()) },
        Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("rare β".into()) },
        Query { start_ns: i64::MAX, end_ns: i64::MIN, tenant: None, token: None },
    ];
    let mut cases = 0;
    for len in 0..=17 {
        let rows: Vec<_> = (0..len).map(row).collect();
        for size in [1, 2, 3, 5, 20] {
            let snapshot = SealedSnapshot::new((0..len).map(row).collect(), NonZeroUsize::new(size).unwrap(), 77, None).unwrap();
            let count = snapshot.anchor().block_count;
            for q in &queries {
                for mask in 0..(1usize << count.min(6)) {
                    let available: Vec<_> = (0..count).map(|ord| ord >= 6 || mask & (1 << ord) != 0).collect();
                    let answer = snapshot.query(q, &available).unwrap();
                    let mut expected_positions = vec![];
                    let mut unavailable = vec![];
                    for (ord, block) in rows.chunks(size).enumerate() {
                        let min = block.iter().map(|r| r.event_time.0).min().unwrap();
                        let max = block.iter().map(|r| r.event_time.0).max().unwrap();
                        let can = q.start_ns <= q.end_ns && max >= q.start_ns && min <= q.end_ns
                            && q.tenant.map_or(true, |t| block.iter().any(|r| r.tenant.0 == t))
                            && q.token.as_ref().map_or(true, |t| block.iter().any(|r| matches!(&r.payload, Payload::Log { body } if body.split_whitespace().any(|word| word == t))));
                        if !can { assert!(matches!(answer.receipt.blocks[ord].disposition, Disposition::Excluded(_))); }
                        else if !available[ord] { unavailable.push(ord); assert!(matches!(answer.receipt.blocks[ord].disposition, Disposition::Unavailable)); }
                        else { assert!(matches!(answer.receipt.blocks[ord].disposition, Disposition::Scanned));
                            for (offset, r) in block.iter().enumerate() { if exact(r, q) { expected_positions.push(ord * size + offset); } }
                        }
                    }
                    assert_eq!(answer.positions, expected_positions, "len={len} size={size} mask={mask} query={q:?}");
                    let expected = if unavailable.is_empty() { CoverageStatus::Complete } else { CoverageStatus::Incomplete { unavailable } };
                    assert_eq!(coverage::verify(snapshot.anchor(), q, &answer.receipt), Ok(expected));
                    if count > 0 {
                        let mut bad = answer.receipt.clone();
                        bad.blocks[0].commitment.summary_digest[0] ^= 1;
                        assert_eq!(coverage::verify(snapshot.anchor(), q, &bad), Err(CoverageError::InvalidReceipt));
                    }
                    cases += 1;
                }
            }
        }
    }
    println!("coverage-gpt-probe: {cases} cases passed");
}
