//! Independent adversarial probes, written without reading candidate internals
//! beyond the frozen public API, to challenge Merkle proof depth/odd-node
//! handling, root binding, zero-block cases, and canonical summary encoding.
use std::collections::BTreeSet;
use std::num::NonZeroUsize;

use fabric_o11y::{Event, EventId, EventTime, ObservedTime, Payload, ResourceId, SourceId, TenantId};
use storage_probe::Query;
use storage_probe::coverage::{self, CoverageStatus, Disposition, SealedSnapshot, Summary};

fn event(id: u64, tenant: u64, time: i64, body: &str) -> Event {
    Event {
        id: EventId(id),
        tenant: TenantId(tenant),
        source: SourceId(1),
        resource: ResourceId(1),
        event_time: EventTime(time),
        observed_time: ObservedTime(time),
        attributes: vec![],
        payload: Payload::Log { body: body.into() },
    }
}

fn any_query() -> Query {
    Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: None }
}

#[test]
fn deep_tree_interior_sibling_flip_is_caught_for_every_leaf() {
    // 13 leaves forces multiple odd-width levels and several tree depths.
    let rows: Vec<_> = (0..13).map(|i| event(i, 1, i as i64, "word")).collect();
    let snapshot = SealedSnapshot::new(rows, NonZeroUsize::new(1).unwrap(), 42, None).unwrap();
    let q = any_query();
    let answer = snapshot.query(&q, &vec![true; 13]).unwrap();
    assert_eq!(
        coverage::verify(snapshot.anchor(), &q, &answer.receipt),
        Ok(CoverageStatus::Complete)
    );
    for leaf in 0..13 {
        for depth in 0..answer.receipt.blocks[leaf].proof.len() {
            let mut bad = answer.receipt.clone();
            bad.blocks[leaf].proof[depth][0] ^= 1;
            assert!(
                coverage::verify(snapshot.anchor(), &q, &bad).is_err(),
                "leaf {leaf} depth {depth} sibling flip must be rejected"
            );
        }
    }
}

#[test]
fn duplicated_odd_leaf_cannot_be_forged_by_swapping_in_a_different_leaf_digest() {
    // Odd leaf count (5) creates a duplicated-last-node level. Confirm the
    // duplicate rule is bound to the actual leaf, not forgeable by substituting
    // another block's own commitment digest as a false "sibling".
    let rows: Vec<_> = (0..5).map(|i| event(i, 1, i as i64, "word")).collect();
    let snapshot = SealedSnapshot::new(rows, NonZeroUsize::new(1).unwrap(), 42, None).unwrap();
    let q = any_query();
    let answer = snapshot.query(&q, &vec![true; 5]).unwrap();
    // Last leaf (index 4) is the duplicated one at the base level.
    let last = 4;
    let other_leaf_proof_first_sibling = answer.receipt.blocks[0].proof[0];
    let mut bad = answer.receipt.clone();
    if !bad.blocks[last].proof.is_empty() {
        bad.blocks[last].proof[0] = other_leaf_proof_first_sibling;
        assert!(
            coverage::verify(snapshot.anchor(), &q, &bad).is_err(),
            "substituting a foreign sibling digest at the duplicated leaf must fail"
        );
    }
}

#[test]
fn proof_depth_matches_expected_for_range_of_block_counts() {
    // proof length must track ceil(log2(count)) exactly; too-short or too-long
    // proofs (already partly covered upstream) are re-checked here across a
    // wider range including powers of two and one-past-power-of-two.
    for count in [1usize, 2, 3, 4, 5, 7, 8, 9, 16, 17, 31, 32] {
        let rows: Vec<_> = (0..count as u64).map(|i| event(i, 1, i as i64, "w")).collect();
        let snapshot = SealedSnapshot::new(rows, NonZeroUsize::new(1).unwrap(), 5, None).unwrap();
        let q = any_query();
        let answer = snapshot.query(&q, &vec![true; count]).unwrap();
        assert_eq!(
            coverage::verify(snapshot.anchor(), &q, &answer.receipt),
            Ok(CoverageStatus::Complete),
            "count={count}"
        );
        let mut expected_depth = 0usize;
        let mut c = count;
        while c > 1 {
            c = c.div_ceil(2);
            expected_depth += 1;
        }
        for block in &answer.receipt.blocks {
            assert_eq!(block.proof.len(), expected_depth, "count={count}");
        }
        // Truncating any one proof must be rejected.
        if expected_depth > 0 {
            let mut bad = answer.receipt.clone();
            bad.blocks[0].proof.pop();
            assert!(coverage::verify(snapshot.anchor(), &q, &bad).is_err(), "count={count}");
        }
    }
}

#[test]
fn zero_block_anchor_rejects_nonzero_block_count_claim_and_wrong_empty_root() {
    let snapshot = SealedSnapshot::new(vec![], NonZeroUsize::new(4).unwrap(), 9, None).unwrap();
    let q = any_query();
    let answer = snapshot.query(&q, &[]).unwrap();
    assert_eq!(
        coverage::verify(snapshot.anchor(), &q, &answer.receipt),
        Ok(CoverageStatus::Complete)
    );
    // A forged anchor claiming one block over an empty receipt must fail layout check.
    let mut forged = snapshot.anchor().clone();
    forged.block_count = 1;
    assert!(coverage::verify(&forged, &q, &answer.receipt).is_err());
    // A forged anchor claiming nonzero rows with zero blocks must fail.
    let mut forged = snapshot.anchor().clone();
    forged.row_count = 1;
    assert!(coverage::verify(&forged, &q, &answer.receipt).is_err());
}

#[test]
fn summary_token_set_boundary_is_not_ambiguous_under_concatenation() {
    // {"a","bc"} vs {"ab","c"}: naive unprefixed concatenation of the sorted
    // sets would both yield "abc". Confirm the digest distinguishes them.
    let a = Summary {
        min_time: 0,
        max_time: 0,
        tenants: BTreeSet::from([1]),
        tokens: BTreeSet::from(["a".to_string(), "bc".to_string()]),
    };
    let b = Summary {
        min_time: 0,
        max_time: 0,
        tenants: BTreeSet::from([1]),
        tokens: BTreeSet::from(["ab".to_string(), "c".to_string()]),
    };
    assert_ne!(coverage::summary_digest(&a), coverage::summary_digest(&b));
}

#[test]
fn excluded_disposition_with_correct_digest_but_wrong_excludes_predicate_is_rejected() {
    // Summary genuinely matches its own digest and time/tenant bounds, but the
    // query's token IS present in the summary's token set, so excludes() must
    // be false; a receipt lying about exclusion should be rejected.
    let rows = vec![event(1, 1, 0, "present")];
    let snapshot = SealedSnapshot::new(rows, NonZeroUsize::new(1).unwrap(), 1, None).unwrap();
    let q = Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("present".into()) };
    let mut answer = snapshot.query(&q, &[true]).unwrap();
    assert!(matches!(answer.receipt.blocks[0].disposition, Disposition::Scanned));
    let real_summary = coverage::summarize(&[event(1, 1, 0, "present")]).unwrap();
    answer.receipt.blocks[0].disposition = Disposition::Excluded(real_summary.clone());
    let mut bad = answer.receipt.clone();
    bad.blocks[0].commitment.summary_digest = coverage::summary_digest(&real_summary);
    assert!(
        coverage::verify(snapshot.anchor(), &q, &bad).is_err(),
        "digest matches but excludes() is false for a present token; must not verify as excluded"
    );
}
