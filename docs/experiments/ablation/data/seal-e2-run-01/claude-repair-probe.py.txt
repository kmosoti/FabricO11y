"""Independent adversarial probe for tools/seal-probe/candidate.py (E2R).

Deliberately does NOT import oracle_support.py or test_oracle.py: this is an
independently re-derived encoder built only from CONTRACT.md's prose, used to
cross-check the candidate from a second, separately-written source of truth.
No repository files are modified. Writes only live under /tmp/fabric-seal-review.

This revision adds targeted-repair checks for the append_logical prefix /
group_id validation fix (duplicate id, gap id, zero tail, corrupt prefix,
valid control, chained appends, external-witness division of responsibility)
on top of the prior general adversarial checks.
"""
import inspect
import importlib.util
import random
import struct
import sys
from hashlib import sha256
from pathlib import Path

CANDIDATE_PATH = Path("/home/kmosoti/projects/fabric_o11y/tools/seal-probe/candidate.py")
DOMAIN = b"Fabric-E2R-seal-v1\0"
assert DOMAIN[-1:] == b"\0" and DOMAIN[:-1] == b"Fabric-E2R-seal-v1"


def load_candidate():
    spec = importlib.util.spec_from_file_location("seal_candidate_probe", CANDIDATE_PATH)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


# ---- independently re-derived encoder (not copied from oracle_support.py) ----

def enc_descriptor(group_id, lengths, sector):
    body = (b"FGD1" + struct.pack("<Q", group_id) + struct.pack("<I", len(lengths))
            + struct.pack("<6I", *(tuple(lengths) + (0,) * (6 - len(lengths)))))
    return body + bytes(sector - len(body))


def enc_frame(index, body, sector):
    raw = b"FGF1" + struct.pack("<I", index) + struct.pack("<I", len(body)) + body
    pad = (-len(raw)) % sector
    return raw + bytes(pad)


def enc_seal(group_id, frame_count, descriptor, frame_sectors, sector, domain=DOMAIN, corrupt_digest=False):
    head = b"FGS1" + struct.pack("<Q", group_id) + struct.pack("<I", frame_count)
    covered = descriptor + b"".join(frame_sectors)
    digest = sha256(domain + head + covered).digest()
    if corrupt_digest:
        digest = bytes([digest[0] ^ 1]) + digest[1:]
    body = head + digest
    return body + bytes(sector - len(body))


def enc_group(group_id, bodies_, sector):
    lengths = [len(b) for b in bodies_]
    descriptor = enc_descriptor(group_id, lengths, sector)
    frame_sectors = [enc_frame(i, b, sector) for i, b in enumerate(bodies_)]
    seal = enc_seal(group_id, len(bodies_), descriptor, frame_sectors, sector)
    return descriptor + b"".join(frame_sectors) + seal


def enc_image(groups, sector):
    return b"".join(enc_group(gid, bodies_, sector) for gid, bodies_ in enumerate(groups))


def rng_body(n, seed):
    r = random.Random(seed)
    return bytes(r.randrange(256) for _ in range(n))


failures = []
checks = 0


def check(name, cond, detail=""):
    global checks
    checks += 1
    if not cond:
        failures.append((name, detail))


def raises_value_error(fn):
    try:
        fn()
    except ValueError:
        return True
    except Exception as exc:  # noqa: BLE001
        return f"wrong exception type: {exc!r}"
    else:
        return False


# ---------------------------------------------------------------------------
impl = load_candidate()
digest = sha256(CANDIDATE_PATH.read_bytes()).hexdigest()
print(f"candidate.py sha256={digest}")

# === General adversarial checks (carried over from prior review round) =====

for sector in (512, 4096):
    groups = (
        (rng_body(57, 101),),
        (rng_body(57, 101), rng_body(4096, 102), rng_body(509, 103)),
    )
    mine = enc_image(groups, sector)
    theirs = impl.encode_groups(groups, sector)
    check(f"cross_encoder_matches[{sector}]", mine == theirs,
          f"len mine={len(mine)} theirs={len(theirs)}")
    result = impl.recover(mine, sector)
    check(f"cross_encoded_image_accepted[{sector}]", result == ("ok", groups, len(mine)))

for sector in (512, 4096):
    prior_bodies = (rng_body(57, 201),)
    inflight_bodies = (rng_body(509, 202), rng_body(1023, 203))
    prior = enc_group(0, prior_bodies, sector)
    inflight = enc_group(1, inflight_bodies, sector)
    clean = prior + inflight
    descriptor_offset = len(prior)
    corrupted = bytearray(clean)
    original_len = struct.unpack_from("<I", corrupted, descriptor_offset + 16)[0]
    new_len = 4096 if original_len != 4096 else 57
    struct.pack_into("<I", corrupted, descriptor_offset + 16, new_len)
    corrupted = bytes(corrupted)
    result = impl.recover(corrupted, sector)
    check(f"corrupted_length_not_treated_as_absent[{sector}]",
          result == ("error", corrupted), f"got {result[0]!r}")
    check(f"corrupted_length_does_not_report_zero_groups[{sector}]",
          not (result[0] == "ok" and result[1] == tuple()), f"{result!r}")

for sector in (512, 4096):
    bodies_ = (rng_body(57, 501),)
    image = enc_group(0, bodies_, sector) + b"\x00" * 17
    result = impl.recover(image, sector)
    check(f"misaligned_length_errors[{sector}]", result == ("error", image), f"got {result[0]!r}")

rng = random.Random(999)
random_ok_with_groups = 0
random_exceptions = []
for trial in range(2000):
    sector = 512 if trial % 2 == 0 else 4096
    n_sectors = 1 + (trial % 9)
    image = bytes(rng.randrange(256) for _ in range(n_sectors * sector))
    try:
        result = impl.recover(image, sector)
    except Exception as exc:  # noqa: BLE001
        random_exceptions.append((trial, repr(exc)))
        continue
    if result[0] == "ok" and result[1]:
        random_ok_with_groups += 1
check("random_corpus_no_group_from_garbage", random_ok_with_groups == 0,
      f"{random_ok_with_groups} random images produced nonempty committed groups")
check("random_corpus_raised_no_exceptions", not random_exceptions, str(random_exceptions[:5]))

# === Targeted-repair checks: append_logical prefix / group_id validation ===

sector = 512
group0 = enc_group(0, (rng_body(57, 11),), sector)  # one ACKed group
group1_bodies = (rng_body(127, 12),)

r = raises_value_error(lambda: impl.append_logical(group0, 0, group1_bodies, sector, None))
check("duplicate_id_rejected", r is True, f"got {r!r}")

r = raises_value_error(lambda: impl.append_logical(group0, 2, group1_bodies, sector, None))
check("gap_id_rejected", r is True, f"got {r!r}")

r = raises_value_error(lambda: impl.append_logical(group0, 999, group1_bodies, sector, None))
check("large_gap_id_rejected", r is True, f"got {r!r}")

zero_tail_previous = group0 + bytes(sector)
r = raises_value_error(lambda: impl.append_logical(zero_tail_previous, 1, group1_bodies, sector, None))
check("zero_tail_previous_rejected", r is True, f"got {r!r}")

r = raises_value_error(lambda: impl.append_logical(bytes(sector), 0, group1_bodies, sector, None))
check("pure_zero_tail_wrong_length_rejected", r is True, f"got {r!r}")

ok_first = impl.append_logical(b"", 0, group1_bodies, sector, None)
check("empty_previous_group_zero_accepted", ok_first[1] is True, f"got {ok_first!r}")
check("empty_previous_recovers_ok", impl.recover(ok_first[0], sector)[0] == "ok")

corrupt_prefix = bytearray(group0)
corrupt_prefix[sector + 12] ^= 1  # inside the frame body region
corrupt_prefix = bytes(corrupt_prefix)
r = raises_value_error(lambda: impl.append_logical(corrupt_prefix, 1, group1_bodies, sector, None))
check("corrupt_prefix_rejected", r is True, f"got {r!r}")

valid = impl.append_logical(group0, 1, group1_bodies, sector, None)
check("valid_control_acked", valid[1] is True, f"got {valid!r}")
scan_after_1 = impl.recover(valid[0], sector)
check("valid_control_scans_ok_two_groups",
      scan_after_1[0] == "ok" and len(scan_after_1[1]) == 2 and scan_after_1[2] == len(valid[0]))

group2_bodies = (rng_body(4096, 13), rng_body(509, 14))
chained = impl.append_logical(valid[0], 2, group2_bodies, sector, None)
check("chained_third_append_acked", chained[1] is True, f"got {chained!r}")
scan_after_2 = impl.recover(chained[0], sector)
check("chained_scan_ok_three_groups",
      scan_after_2[0] == "ok" and len(scan_after_2[1]) == 3 and scan_after_2[2] == len(chained[0]))

r = raises_value_error(lambda: impl.append_logical(chained[0], 2, group2_bodies, sector, None))
check("duplicate_id_at_new_tip_rejected", r is True, f"got {r!r}")

sig = inspect.signature(impl.append_logical)
check("append_logical_signature_unchanged",
      list(sig.parameters) == ["previous", "group_id", "bodies", "sector_size", "fail_at"],
      str(list(sig.parameters)))
supervise_sig = inspect.signature(impl.supervise)
check("supervise_still_takes_external_witness",
      list(supervise_sig.parameters) == ["external_io_error", "recovery_result"],
      str(list(supervise_sig.parameters)))
poisoned_bytes_ok_scan = impl.recover(group0, sector)
check("poisoned_witness_blocks_despite_ok_scan",
      impl.supervise(True, poisoned_bytes_ok_scan) == "rebuild_from_trusted_source")

# Regression parity: replay the exact (previous, group_id) pairs from
# docs/experiments/ablation/data/seal-e2-run-01/pre-repair-regression.json,
# independently re-derived (own encoder, not oracle_support.bodies/group_bytes).
prior_for_regression = enc_group(0, (rng_body(57, 11),), 512)
next_bodies = (rng_body(57, 12),)
pairs = [
    (prior_for_regression, 0),
    (prior_for_regression, 2),
    (prior_for_regression + bytes(512), 1),
    (bytes(512), 0),
]
for previous, group_id in pairs:
    r = raises_value_error(lambda p=previous, g=group_id: impl.append_logical(p, g, next_bodies, 512, None))
    check(f"regression_pair_rejected[len={len(previous)},id={group_id}]", r is True, f"got {r!r}")
bad_regression = bytearray(prior_for_regression)
bad_regression[512 + 12] ^= 1
r = raises_value_error(lambda: impl.append_logical(bytes(bad_regression), 1, next_bodies, 512, None))
check("regression_bitflip_rejected", r is True, f"got {r!r}")
valid_regression = impl.append_logical(prior_for_regression, 1, next_bodies, 512, None)
check("regression_valid_control_acked", valid_regression[1] is True)
check("regression_valid_control_scans_ok", impl.recover(valid_regression[0], 512)[0] == "ok")

# ---------------------------------------------------------------------------
print(f"checks={checks} failures={len(failures)}")
for name, detail in failures:
    print(f"FAIL {name}: {detail}")
sys.exit(1 if failures else 0)
