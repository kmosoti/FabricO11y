from pathlib import Path
import importlib.util
import sys

root=Path('/home/kmosoti/projects/fabric_o11y/tools/seal-probe')
sys.path.insert(0,str(root))
from oracle_support import bodies, group_bytes
spec=importlib.util.spec_from_file_location('candidate_repair',root/'candidate.py')
model=importlib.util.module_from_spec(spec)
spec.loader.exec_module(model)
checks=0
for sector in (512,4096):
    prior=b''.join(group_bytes(i,bodies(1,11+i),sector) for i in range(2))
    good_bodies=bodies(2,13)
    invalid=[
        (prior,0,'duplicate_zero'),
        (prior,1,'duplicate_last'),
        (prior,3,'gap'),
        (prior+bytes(sector),2,'zero_tail'),
        (prior[:-sector]+bytes(sector),2,'erased_seal'),
        (bytes(sector)+prior,2,'zero_head'),
    ]
    changed=bytearray(prior)
    changed[sector+12] ^= 1
    invalid.append((bytes(changed),2,'corrupt_frame'))
    for image,group_id,label in invalid:
        before=image
        try:
            model.append_logical(image,group_id,good_bodies,sector,None)
        except ValueError:
            assert image==before
        else:
            raise AssertionError((sector,label))
        checks+=1
    for fail_at in (None,'sync'):
        result=model.append_logical(prior,2,good_bodies,sector,fail_at)
        assert result[0].startswith(prior)
        assert model.recover(result[0],sector)==('ok',(bodies(1,11),bodies(1,12),good_bodies),len(result[0]))
        if fail_at is None:
            assert result[1:4]==(True,False,False)
        else:
            assert result[1:4]==(False,True,True)
        checks+=1
    empty=model.append_logical(b'',0,good_bodies,sector,None)
    assert empty[1] is True
    assert model.recover(empty[0],sector)==('ok',(good_bodies,),len(empty[0]))
    checks+=1
print(f'PASS: {checks} independent repair cases')
