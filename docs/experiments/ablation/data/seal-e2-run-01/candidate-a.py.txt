"""Independent pure model for the frozen E2R group-seal framing."""

from __future__ import annotations

import hashlib


_DOMAIN = b"Fabric-E2R-seal-v1\0"
_MAGIC_DESCRIPTOR = b"FGD1"
_MAGIC_FRAME = b"FGF1"
_MAGIC_SEAL = b"FGS1"


def _sector_size(value: int) -> int:
    if type(value) is not int or value not in (512, 4096):
        raise ValueError("sector_size must be 512 or 4096")
    return value


def _valid_bodies(bodies: tuple[bytes, ...]) -> tuple[bytes, ...]:
    if not isinstance(bodies, tuple) or not 1 <= len(bodies) <= 6:
        raise ValueError("a group must contain between one and six bodies")
    for body in bodies:
        if not isinstance(body, bytes) or not 57 <= len(body) <= 4096:
            raise ValueError("each body must be bytes with length 57..4096")
    return bodies


def _padded(raw: bytes, sector: int) -> bytes:
    return raw + bytes((-len(raw)) % sector)


def _encode_group(group_id: int, bodies: tuple[bytes, ...], sector: int) -> bytes:
    bodies = _valid_bodies(bodies)
    if type(group_id) is not int or not 0 <= group_id < (1 << 64):
        raise ValueError("group_id must fit in u64")

    descriptor = bytearray(sector)
    descriptor[0:4] = _MAGIC_DESCRIPTOR
    descriptor[4:12] = group_id.to_bytes(8, "little")
    descriptor[12:16] = len(bodies).to_bytes(4, "little")
    for index, body in enumerate(bodies):
        start = 16 + 4 * index
        descriptor[start:start + 4] = len(body).to_bytes(4, "little")

    frames = bytearray()
    for index, body in enumerate(bodies):
        frame = (_MAGIC_FRAME + index.to_bytes(4, "little") +
                 len(body).to_bytes(4, "little") + body)
        frames.extend(_padded(frame, sector))

    seal = bytearray(sector)
    seal[0:4] = _MAGIC_SEAL
    seal[4:12] = group_id.to_bytes(8, "little")
    seal[12:16] = len(bodies).to_bytes(4, "little")
    seal[16:48] = hashlib.sha256(_DOMAIN + bytes(seal[:16]) +
                                 bytes(descriptor) + bytes(frames)).digest()
    return bytes(descriptor) + bytes(frames) + bytes(seal)


def encode_groups(groups: tuple[tuple[bytes, ...], ...], sector_size: int) -> bytes:
    sector = _sector_size(sector_size)
    if not isinstance(groups, tuple):
        raise ValueError("groups must be a tuple")
    return b"".join(_encode_group(group_id, _valid_bodies(group), sector)
                    for group_id, group in enumerate(groups))


def _all_zero(data: bytes) -> bool:
    return not any(data)


def _well_formed_seal(seal: bytes, sector: int) -> bool:
    return (len(seal) == sector and seal[:4] == _MAGIC_SEAL and
            _all_zero(seal[48:]))


def recover(image: bytes, sector_size: int) -> tuple:
    sector = _sector_size(sector_size)
    if not isinstance(image, bytes) or len(image) % sector:
        raise ValueError("image must be sector-aligned bytes")

    offset = 0
    expected_id = 0
    committed: list[tuple[bytes, ...]] = []
    total = len(image)

    while offset < total:
        descriptor = image[offset:offset + sector]
        if len(descriptor) < sector:
            return ("error", image)

        # A missing descriptor cannot locate its seal. Recognize a complete
        # seal-shaped sector anywhere in this tail and fail closed; otherwise
        # the tail is uncommitted. Nonzero malformed data is also conservatively
        # rejected so a damaged earlier group can never be silently skipped.
        if descriptor[:4] != _MAGIC_DESCRIPTOR:
            tail = image[offset:]
            if _all_zero(tail):
                return ("ok", tuple(committed), offset)
            for pos in range(0, len(tail), sector):
                candidate = tail[pos:pos + sector]
                if _well_formed_seal(candidate, sector):
                    return ("error", image)
            return ("error", image)

        group_id = int.from_bytes(descriptor[4:12], "little")
        count = int.from_bytes(descriptor[12:16], "little")
        if (group_id != expected_id or not 1 <= count <= 6 or
                not _all_zero(descriptor[40:])):
            return ("error", image)
        lengths = [int.from_bytes(descriptor[16 + 4 * i:20 + 4 * i], "little")
                   for i in range(6)]
        if (any(not 57 <= length <= 4096 for length in lengths[:count]) or
                any(lengths[count:])):
            return ("error", image)

        frame_offset = offset + sector
        frame_parts: list[bytes] = []
        decoded_bodies: list[bytes] = []
        framing_valid = True
        for index, length in enumerate(lengths[:count]):
            span = ((12 + length + sector - 1) // sector) * sector
            part = image[frame_offset:frame_offset + span]
            if len(part) != span:
                framing_valid = False
                part = part + bytes(span - len(part))
            frame_parts.append(part)
            header = _MAGIC_FRAME + index.to_bytes(4, "little") + length.to_bytes(4, "little")
            if (part[:12] != header or
                    not _all_zero(part[12 + length:])):
                framing_valid = False
            decoded_bodies.append(part[12:12 + length])
            frame_offset += span

        seal_offset = frame_offset
        seal = image[seal_offset:seal_offset + sector]
        if len(seal) != sector:
            seal = seal + bytes(sector - len(seal))
        has_seal = not _all_zero(seal)

        if not has_seal:
            # No commit marker: this final group is wholly discarded, even if
            # some of its descriptor/frame sectors reached the image.
            return ("ok", tuple(committed), offset)

        # A nonzero seal sector is a commit claim. Every field, padding byte,
        # frame, and covered byte must agree or the whole image is rejected.
        if (not _well_formed_seal(seal, sector) or
                int.from_bytes(seal[4:12], "little") != group_id or
                int.from_bytes(seal[12:16], "little") != count):
            return ("error", image)
        raw_frames = b"".join(frame_parts)
        digest = hashlib.sha256(_DOMAIN + seal[:16] + descriptor + raw_frames).digest()
        if seal[16:48] != digest or not framing_valid:
            return ("error", image)

        committed.append(tuple(decoded_bodies))
        offset = seal_offset + sector
        expected_id += 1

    return ("ok", tuple(committed), offset)


def supervise(external_io_error: bool, recovery_result: tuple) -> str:
    if type(external_io_error) is not bool:
        raise ValueError("external_io_error must be bool")
    if (not external_io_error and isinstance(recovery_result, tuple) and
            len(recovery_result) == 3 and recovery_result[0] == "ok"):
        return "resume_allowed"
    return "rebuild_from_trusted_source"


def append_logical(previous: bytes, group_id: int, bodies: tuple[bytes, ...],
                   sector_size: int, fail_at: str | None) -> tuple:
    sector = _sector_size(sector_size)
    if not isinstance(previous, bytes) or len(previous) % sector:
        raise ValueError("previous must be sector-aligned bytes")
    group = _encode_group(group_id, bodies, sector)
    writes = tuple(f"write:{i}" for i in range(len(group) // sector))
    operations = writes + ("sync",)
    if fail_at is not None and fail_at not in operations:
        raise ValueError("fail_at must name an operation or be None")

    if fail_at is None:
        return (previous + group, True, False, False, operations)
    if fail_at == "sync":
        visible = group
    else:
        failed_index = int(fail_at.split(":", 1)[1])
        written = (failed_index + 1) * sector
        visible = group[:written] + bytes(len(group) - written)
    return (previous + visible, False, True, True, operations)
