//! Immutable JSON block snapshots and digest-bound query checkpoints.
//!
//! This research format uses the coverage builder for its trusted anchor. Raw
//! files remain untrusted at read time and are checked against that anchor.

use std::{
    fs::{self, File, OpenOptions},
    io::{self, Read, Write},
    num::NonZeroUsize,
    path::{Path, PathBuf},
    sync::atomic::{AtomicU64, Ordering},
};

use fabric_o11y::{
    Attribute, Event, EventId, EventTime, ObservedTime, Payload, ResourceId, Scalar, SourceId,
    TenantId,
};
use serde::{Deserialize, Serialize};
use sha2::{Digest as _, Sha256};

use crate::{
    Query,
    coverage::{self, Anchor, BlockCommitment, BlockReceipt, Disposition, Receipt, Summary},
    resume::{self, Accumulator, Binding, MatchedRow, Page, Residual},
};

const VERSION: u32 = 1;
const MAX_FILE_BYTES: usize = 64 * 1024 * 1024;
static NEXT_STAGE: AtomicU64 = AtomicU64::new(0);

#[derive(Clone, Debug, Eq, PartialEq)]
pub struct Publication {
    pub anchor: Anchor,
    pub block_rows: usize,
    pub codec_version: u32,
}

#[derive(Clone, Debug, Default, Eq, PartialEq)]
pub struct ReadMetrics {
    pub metadata_bytes: u64,
    pub raw_bytes: u64,
    pub raw_files: usize,
    pub unavailable: Vec<usize>,
}

#[derive(Clone, Debug, Eq, PartialEq)]
pub struct DiskAnswer {
    pub page: Page,
    pub reads: ReadMetrics,
}

#[derive(Clone, Debug)]
pub struct DiskSnapshot {
    dir: PathBuf,
    publication: Publication,
    metadata: Manifest,
    metadata_bytes: u64,
}

#[derive(Deserialize)]
#[serde(deny_unknown_fields)]
struct EventEnvelope {
    version: u32,
    events: Vec<EventWire>,
}

#[derive(Deserialize)]
#[serde(deny_unknown_fields)]
struct EventWire {
    id: u64,
    tenant: u64,
    source: u64,
    resource: u64,
    event_time: i64,
    observed_time: i64,
    attributes: Vec<AttributeWire>,
    payload: PayloadWire,
}

#[derive(Deserialize)]
#[serde(deny_unknown_fields)]
struct AttributeWire {
    key: String,
    value: ScalarWire,
}

#[derive(Deserialize)]
#[serde(tag = "kind", content = "value", deny_unknown_fields)]
enum ScalarWire {
    #[serde(rename = "bool")]
    Bool(bool),
    #[serde(rename = "i64")]
    I64(i64),
    #[serde(rename = "u64")]
    U64(u64),
    #[serde(rename = "string")]
    String(String),
    #[serde(rename = "f64_bits")]
    F64Bits(String),
}

#[derive(Deserialize)]
#[serde(tag = "kind", deny_unknown_fields)]
enum PayloadWire {
    #[serde(rename = "log")]
    Log { body: String },
    #[serde(rename = "gauge")]
    Gauge {
        name: String,
        value_bits: String,
        unit: String,
    },
}

#[derive(Serialize)]
struct EventEnvelopeRef<'a> {
    version: u32,
    events: Vec<EventWireRef<'a>>,
}

#[derive(Serialize)]
struct EventWireRef<'a> {
    id: u64,
    tenant: u64,
    source: u64,
    resource: u64,
    event_time: i64,
    observed_time: i64,
    attributes: Vec<AttributeWireRef<'a>>,
    payload: PayloadWireRef<'a>,
}

#[derive(Serialize)]
struct AttributeWireRef<'a> {
    key: &'a str,
    value: ScalarWireRef<'a>,
}

#[derive(Serialize)]
#[serde(tag = "kind", content = "value")]
enum ScalarWireRef<'a> {
    #[serde(rename = "bool")]
    Bool(bool),
    #[serde(rename = "i64")]
    I64(i64),
    #[serde(rename = "u64")]
    U64(u64),
    #[serde(rename = "string")]
    String(&'a str),
    #[serde(rename = "f64_bits")]
    F64Bits(String),
}

#[derive(Serialize)]
#[serde(tag = "kind")]
enum PayloadWireRef<'a> {
    #[serde(rename = "log")]
    Log { body: &'a str },
    #[serde(rename = "gauge")]
    Gauge {
        name: &'a str,
        value_bits: String,
        unit: &'a str,
    },
}

#[derive(Serialize, Deserialize)]
#[serde(deny_unknown_fields)]
struct PublicationEnvelope {
    version: u32,
    publication: PublicationWire,
}

#[derive(Serialize, Deserialize)]
#[serde(deny_unknown_fields)]
struct PublicationWire {
    anchor: Anchor,
    block_rows: usize,
    codec_version: u32,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
#[serde(deny_unknown_fields)]
struct Manifest {
    version: u32,
    block_rows: usize,
    anchor: Anchor,
    blocks: Vec<BlockMetadata>,
}

#[derive(Clone, Debug, Serialize, Deserialize)]
#[serde(deny_unknown_fields)]
struct BlockMetadata {
    commitment: BlockCommitment,
    summary: Summary,
    proof: Vec<coverage::Digest>,
}

#[derive(Serialize, Deserialize)]
#[serde(deny_unknown_fields)]
struct CheckpointEnvelope {
    version: u32,
    binding: Binding,
    pages: Vec<Page>,
}

pub fn encode_events(events: &[Event]) -> io::Result<Vec<u8>> {
    let wire = EventEnvelopeRef {
        version: VERSION,
        events: events.iter().map(event_to_wire_ref).collect(),
    };
    bounded_json(&wire)
}

pub fn decode_events(bytes: &[u8]) -> io::Result<Vec<Event>> {
    if bytes.len() > MAX_FILE_BYTES {
        return Err(invalid_input("event file exceeds 64 MiB"));
    }
    let wire: EventEnvelope = serde_json::from_slice(bytes).map_err(invalid_data)?;
    if wire.version != VERSION {
        return Err(invalid_input("unsupported event codec version"));
    }
    wire.events.into_iter().map(event_from_wire).collect()
}

fn event_to_wire_ref(event: &Event) -> EventWireRef<'_> {
    EventWireRef {
        id: event.id.0,
        tenant: event.tenant.0,
        source: event.source.0,
        resource: event.resource.0,
        event_time: event.event_time.0,
        observed_time: event.observed_time.0,
        attributes: event
            .attributes
            .iter()
            .map(|attribute| AttributeWireRef {
                key: &attribute.key,
                value: match &attribute.value {
                    Scalar::Bool(value) => ScalarWireRef::Bool(*value),
                    Scalar::I64(value) => ScalarWireRef::I64(*value),
                    Scalar::U64(value) => ScalarWireRef::U64(*value),
                    Scalar::String(value) => ScalarWireRef::String(value),
                    Scalar::F64(value) => ScalarWireRef::F64Bits(hex64(value.to_bits())),
                },
            })
            .collect(),
        payload: match &event.payload {
            Payload::Log { body } => PayloadWireRef::Log { body },
            Payload::Gauge { name, value, unit } => PayloadWireRef::Gauge {
                name,
                value_bits: hex64(value.to_bits()),
                unit,
            },
        },
    }
}

fn event_from_wire(wire: EventWire) -> io::Result<Event> {
    let attributes = wire
        .attributes
        .into_iter()
        .map(|attribute| {
            Ok(Attribute {
                key: attribute.key,
                value: match attribute.value {
                    ScalarWire::Bool(value) => Scalar::Bool(value),
                    ScalarWire::I64(value) => Scalar::I64(value),
                    ScalarWire::U64(value) => Scalar::U64(value),
                    ScalarWire::String(value) => Scalar::String(value),
                    ScalarWire::F64Bits(value) => Scalar::F64(f64::from_bits(parse_hex64(&value)?)),
                },
            })
        })
        .collect::<io::Result<Vec<_>>>()?;
    let payload = match wire.payload {
        PayloadWire::Log { body } => Payload::Log { body },
        PayloadWire::Gauge {
            name,
            value_bits,
            unit,
        } => Payload::Gauge {
            name,
            value: f64::from_bits(parse_hex64(&value_bits)?),
            unit,
        },
    };
    Ok(Event {
        id: EventId(wire.id),
        tenant: TenantId(wire.tenant),
        source: SourceId(wire.source),
        resource: ResourceId(wire.resource),
        event_time: EventTime(wire.event_time),
        observed_time: ObservedTime(wire.observed_time),
        attributes,
        payload,
    })
}

fn hex64(bits: u64) -> String {
    format!("{bits:016x}")
}

fn parse_hex64(value: &str) -> io::Result<u64> {
    if value.len() != 16
        || !value
            .bytes()
            .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte))
    {
        return Err(invalid_input(
            "floating-point bits must be 16 lowercase hexadecimal digits",
        ));
    }
    u64::from_str_radix(value, 16).map_err(|_| invalid_input("invalid floating-point bit encoding"))
}

pub fn publish(
    events: Vec<Event>,
    block_rows: NonZeroUsize,
    snapshot_id: u64,
    final_dir: &Path,
) -> io::Result<Publication> {
    let parent = final_dir
        .parent()
        .filter(|path| !path.as_os_str().is_empty())
        .unwrap_or(Path::new("."));
    if !parent.is_dir() {
        return Err(invalid_input("snapshot parent must already exist"));
    }
    let name = final_dir
        .file_name()
        .ok_or_else(|| invalid_input("snapshot path must name a directory"))?;
    let lock_path = parent.join(format!(".{}.publish.lock", name.to_string_lossy()));
    let lock = OpenOptions::new()
        .write(true)
        .create_new(true)
        .open(&lock_path)?;
    let _lock_guard = LockGuard {
        path: lock_path,
        file: Some(lock),
    };
    if final_dir.exists() {
        return Err(already_exists("snapshot directory already exists"));
    }

    let summaries = events
        .chunks(block_rows.get())
        .map(coverage::summarize)
        .collect::<Result<Vec<_>, _>>()
        .map_err(|_| invalid_input("cannot summarize snapshot rows"))?;
    let mut commitments = Vec::with_capacity(summaries.len());
    let mut start = 0usize;
    for (ordinal, (rows, summary)) in events.chunks(block_rows.get()).zip(&summaries).enumerate() {
        commitments.push(BlockCommitment {
            ordinal,
            start,
            len: rows.len(),
            rows_digest: coverage::rows_digest(rows),
            summary_digest: coverage::summary_digest(summary),
        });
        start = start
            .checked_add(rows.len())
            .ok_or_else(|| invalid_input("snapshot row count overflow"))?;
    }
    let stage = create_stage(parent, name)?;
    for (ordinal, rows) in events.chunks(block_rows.get()).enumerate() {
        write_new_synced(&stage.join(block_name(ordinal)), &encode_events(rows)?)?;
    }
    let sealed = coverage::SealedSnapshot::new(events, block_rows, snapshot_id, None)
        .map_err(|_| invalid_input("trusted snapshot builder rejected rows"))?;
    let (anchor, proofs) = coverage::authenticate(snapshot_id, start, &commitments);
    if &anchor != sealed.anchor() {
        return Err(invalid_data("trusted builder anchor mismatch"));
    }
    let metadata = Manifest {
        version: VERSION,
        block_rows: block_rows.get(),
        anchor: anchor.clone(),
        blocks: commitments
            .into_iter()
            .zip(summaries)
            .zip(proofs)
            .map(|((commitment, summary), proof)| BlockMetadata {
                commitment,
                summary,
                proof,
            })
            .collect(),
    };
    write_new_synced(&stage.join("manifest.json"), &json_bytes(&metadata)?)?;
    sync_dir(&stage)?;
    if final_dir.exists() {
        return Err(already_exists("snapshot directory already exists"));
    }
    fs::rename(&stage, final_dir)?;
    sync_dir(parent)?;
    _lock_guard.release(parent)?;
    let publication = Publication {
        anchor,
        block_rows: block_rows.get(),
        codec_version: VERSION,
    };
    validate_publication(&publication)?;
    Ok(publication)
}

fn create_stage(parent: &Path, name: &std::ffi::OsStr) -> io::Result<PathBuf> {
    for _ in 0..100 {
        let suffix = NEXT_STAGE.fetch_add(1, Ordering::Relaxed);
        let stage = parent.join(format!(
            ".{}.stage-{}-{suffix}",
            name.to_string_lossy(),
            std::process::id()
        ));
        #[cfg(unix)]
        {
            use std::os::unix::fs::DirBuilderExt;
            let mut builder = fs::DirBuilder::new();
            builder.mode(0o700);
            match builder.create(&stage) {
                Ok(()) => return Ok(stage),
                Err(error) if error.kind() == io::ErrorKind::AlreadyExists => continue,
                Err(error) => return Err(error),
            }
        }
        #[cfg(not(unix))]
        match fs::create_dir(&stage) {
            Ok(()) => return Ok(stage),
            Err(error) if error.kind() == io::ErrorKind::AlreadyExists => continue,
            Err(error) => return Err(error),
        }
    }
    Err(io::Error::new(
        io::ErrorKind::AlreadyExists,
        "could not allocate unique staging directory",
    ))
}

fn block_name(ordinal: usize) -> String {
    format!("block-{ordinal}.json")
}

pub fn save_publication(path: &Path, publication: &Publication) -> io::Result<()> {
    validate_publication(publication)?;
    let envelope = PublicationEnvelope {
        version: VERSION,
        publication: PublicationWire {
            anchor: publication.anchor.clone(),
            block_rows: publication.block_rows,
            codec_version: publication.codec_version,
        },
    };
    write_fresh(path, &json_bytes(&envelope)?)
}

pub fn load_publication(path: &Path) -> io::Result<Publication> {
    let bytes = read_bounded(path)?;
    let envelope: PublicationEnvelope = serde_json::from_slice(&bytes).map_err(invalid_data)?;
    if envelope.version != VERSION {
        return Err(invalid_input("unsupported publication version"));
    }
    let publication = Publication {
        anchor: envelope.publication.anchor,
        block_rows: envelope.publication.block_rows,
        codec_version: envelope.publication.codec_version,
    };
    validate_publication(&publication)?;
    Ok(publication)
}

fn validate_publication(publication: &Publication) -> io::Result<()> {
    if publication.codec_version != VERSION || publication.block_rows == 0 {
        return Err(invalid_input("invalid publication version or block size"));
    }
    let expected_blocks = publication
        .anchor
        .row_count
        .div_ceil(publication.block_rows);
    if publication.anchor.block_count != expected_blocks {
        return Err(invalid_input("publication anchor layout is inconsistent"));
    }
    if expected_blocks == 0 {
        let (empty, _) = coverage::authenticate(publication.anchor.snapshot_id, 0, &[]);
        if publication.anchor != empty {
            return Err(invalid_input("invalid empty publication anchor"));
        }
    }
    Ok(())
}

impl DiskSnapshot {
    pub fn open(dir: &Path, expected: Publication) -> io::Result<Self> {
        validate_publication(&expected)?;
        if !dir.is_dir() {
            return Err(invalid_input("snapshot directory does not exist"));
        }
        let mut bytes_read = 0u64;
        let primary_path = dir.join("manifest.json");
        let primary = read_bounded(&primary_path)
            .map(|bytes| {
                bytes_read = bytes_read.saturating_add(bytes.len() as u64);
                bytes
            })
            .and_then(|bytes| parse_manifest(&bytes, &expected));
        let metadata = match primary {
            Ok(metadata) => metadata,
            Err(primary_error) => {
                let rebuilt = read_bounded(&dir.join("manifest.rebuilt.json"))
                    .map(|bytes| {
                        bytes_read = bytes_read.saturating_add(bytes.len() as u64);
                        bytes
                    })
                    .and_then(|bytes| parse_manifest(&bytes, &expected));
                rebuilt.map_err(|_| primary_error)?
            }
        };
        Ok(Self {
            dir: dir.to_path_buf(),
            publication: expected,
            metadata,
            metadata_bytes: bytes_read,
        })
    }

    pub fn query(&self, binding: &Binding, available: &[bool]) -> io::Result<DiskAnswer> {
        self.query_mask(binding, available, None)
    }

    pub fn resume(&self, residual: &Residual, available: &[bool]) -> io::Result<DiskAnswer> {
        self.check_binding(&residual.binding)?;
        self.check_available_len(available)?;
        let mut requested = vec![false; self.metadata.blocks.len()];
        let mut previous = None;
        for commitment in &residual.blocks {
            let ordinal = commitment.ordinal;
            if ordinal >= requested.len()
                || previous.is_some_and(|last| ordinal <= last)
                || self.metadata.blocks[ordinal].commitment != *commitment
            {
                return Err(invalid_input(
                    "residual does not match authenticated layout",
                ));
            }
            requested[ordinal] = true;
            previous = Some(ordinal);
        }
        let permitted: Vec<bool> = requested
            .iter()
            .zip(available)
            .map(|(want, have)| *want && *have)
            .collect();
        self.query_mask(&residual.binding, &permitted, Some(&requested))
    }

    fn query_mask(
        &self,
        binding: &Binding,
        available: &[bool],
        requested: Option<&[bool]>,
    ) -> io::Result<DiskAnswer> {
        self.check_binding(binding)?;
        self.check_available_len(available)?;
        let mut reads = ReadMetrics {
            metadata_bytes: self.metadata_bytes,
            ..ReadMetrics::default()
        };
        let mut blocks = Vec::with_capacity(self.metadata.blocks.len());
        let mut rows = Vec::new();
        for (ordinal, metadata) in self.metadata.blocks.iter().enumerate() {
            let disposition = if excludes(&metadata.summary, &binding.query) {
                Disposition::Excluded(metadata.summary.clone())
            } else if requested.is_some_and(|mask| !mask[ordinal]) || !available[ordinal] {
                Disposition::Unavailable
            } else {
                match self.read_valid_block(ordinal, metadata, &mut reads) {
                    Some(events) => {
                        for (offset, event) in events.iter().enumerate() {
                            if matches_query(event, &binding.query) {
                                rows.push(MatchedRow {
                                    block: ordinal,
                                    offset,
                                    digest: coverage::rows_digest(std::slice::from_ref(event)),
                                });
                            }
                        }
                        Disposition::Scanned
                    }
                    None => {
                        reads.unavailable.push(ordinal);
                        Disposition::Unavailable
                    }
                }
            };
            blocks.push(BlockReceipt {
                commitment: metadata.commitment.clone(),
                proof: metadata.proof.clone(),
                disposition,
            });
        }
        let receipt = Receipt {
            snapshot_id: self.publication.anchor.snapshot_id,
            query: binding.query.clone(),
            blocks,
        };
        Ok(DiskAnswer {
            page: Page {
                binding: binding.clone(),
                receipt,
                rows,
            },
            reads,
        })
    }

    fn read_valid_block(
        &self,
        ordinal: usize,
        metadata: &BlockMetadata,
        reads: &mut ReadMetrics,
    ) -> Option<Vec<Event>> {
        for path in [
            self.dir.join(block_name(ordinal)),
            self.dir.join("cold").join(block_name(ordinal)),
        ] {
            reads.raw_files = reads.raw_files.saturating_add(1);
            let Ok(bytes) = read_bounded(&path) else {
                continue;
            };
            reads.raw_bytes = reads.raw_bytes.saturating_add(bytes.len() as u64);
            let Ok(events) = decode_events(&bytes) else {
                continue;
            };
            if events.len() == metadata.commitment.len
                && coverage::rows_digest(&events) == metadata.commitment.rows_digest
            {
                return Some(events);
            }
        }
        None
    }

    fn check_binding(&self, binding: &Binding) -> io::Result<()> {
        if binding.anchor != self.publication.anchor
            || binding.tokenizer_version != resume::TOKENIZER_VERSION
            || binding.order_version != resume::ORDER_VERSION
        {
            return Err(invalid_input(
                "query binding differs from snapshot or supported semantics",
            ));
        }
        Ok(())
    }

    fn check_available_len(&self, available: &[bool]) -> io::Result<()> {
        if available.len() != self.metadata.blocks.len() {
            return Err(invalid_input(
                "availability length differs from block count",
            ));
        }
        Ok(())
    }
}

fn parse_manifest(bytes: &[u8], expected: &Publication) -> io::Result<Manifest> {
    let manifest: Manifest = serde_json::from_slice(bytes).map_err(invalid_data)?;
    validate_manifest(&manifest, expected)?;
    Ok(manifest)
}

fn validate_manifest(manifest: &Manifest, expected: &Publication) -> io::Result<()> {
    if manifest.version != VERSION
        || manifest.block_rows != expected.block_rows
        || manifest.anchor != expected.anchor
        || manifest.blocks.len() != expected.anchor.block_count
    {
        return Err(invalid_data("manifest does not match trusted publication"));
    }
    let mut next_start = 0usize;
    let mut commitments = Vec::with_capacity(manifest.blocks.len());
    for (ordinal, block) in manifest.blocks.iter().enumerate() {
        let c = &block.commitment;
        let expected_len = expected
            .anchor
            .row_count
            .saturating_sub(next_start)
            .min(expected.block_rows);
        if expected_len == 0
            || c.ordinal != ordinal
            || c.start != next_start
            || c.len != expected_len
            || block.summary.min_time > block.summary.max_time
            || block.summary.tenants.is_empty()
            || coverage::summary_digest(&block.summary) != c.summary_digest
        {
            return Err(invalid_data("manifest block metadata is invalid"));
        }
        next_start = next_start
            .checked_add(c.len)
            .ok_or_else(|| invalid_data("manifest layout overflow"))?;
        commitments.push(c.clone());
    }
    if next_start != expected.anchor.row_count {
        return Err(invalid_data("manifest row layout is incomplete"));
    }
    let (anchor, proofs) = coverage::authenticate(
        expected.anchor.snapshot_id,
        expected.anchor.row_count,
        &commitments,
    );
    if anchor != expected.anchor
        || manifest
            .blocks
            .iter()
            .zip(proofs)
            .any(|(block, proof)| block.proof != proof)
    {
        return Err(invalid_data("manifest proof does not match trusted anchor"));
    }
    // Reuse the coverage verifier for its strict proof/layout checks.
    let receipt = Receipt {
        snapshot_id: manifest.anchor.snapshot_id,
        query: Query {
            start_ns: i64::MIN,
            end_ns: i64::MAX,
            tenant: None,
            token: None,
        },
        blocks: manifest
            .blocks
            .iter()
            .map(|block| BlockReceipt {
                commitment: block.commitment.clone(),
                proof: block.proof.clone(),
                disposition: Disposition::Unavailable,
            })
            .collect(),
    };
    coverage::verify(&manifest.anchor, &receipt.query, &receipt)
        .map_err(|_| invalid_data("manifest coverage proof failed"))?;
    Ok(())
}

pub fn rebuild_manifest(dir: &Path, expected: &Publication) -> io::Result<PathBuf> {
    validate_publication(expected)?;
    if !dir.is_dir() {
        return Err(invalid_input("snapshot directory does not exist"));
    }
    let target = dir.join("manifest.rebuilt.json");
    if target.exists() {
        return Err(already_exists("rebuilt manifest already exists"));
    }
    let block_rows =
        NonZeroUsize::new(expected.block_rows).ok_or_else(|| invalid_input("zero block size"))?;
    let mut events = Vec::with_capacity(expected.anchor.row_count);
    for ordinal in 0..expected.anchor.block_count {
        let expected_len = expected
            .anchor
            .row_count
            .saturating_sub(ordinal * block_rows.get())
            .min(block_rows.get());
        let mut valid_rows = None;
        for path in [
            dir.join(block_name(ordinal)),
            dir.join("cold").join(block_name(ordinal)),
        ] {
            let Ok(bytes) = read_bounded(&path) else {
                continue;
            };
            let Ok(rows) = decode_events(&bytes) else {
                continue;
            };
            if rows.len() == expected_len {
                valid_rows = Some(rows);
                break;
            }
        }
        let rows =
            valid_rows.ok_or_else(|| invalid_data("snapshot block is missing or invalid"))?;
        if rows.len() != expected_len {
            return Err(invalid_data("snapshot block has wrong row count"));
        }
        events.extend(rows);
    }
    if events.len() != expected.anchor.row_count {
        return Err(invalid_data("snapshot row count mismatch"));
    }
    let summaries = events
        .chunks(block_rows.get())
        .map(coverage::summarize)
        .collect::<Result<Vec<_>, _>>()
        .map_err(|_| invalid_data("cannot summarize rebuilt rows"))?;
    let mut commitments = Vec::with_capacity(summaries.len());
    let mut start = 0usize;
    for (ordinal, (rows, summary)) in events.chunks(block_rows.get()).zip(&summaries).enumerate() {
        commitments.push(BlockCommitment {
            ordinal,
            start,
            len: rows.len(),
            rows_digest: coverage::rows_digest(rows),
            summary_digest: coverage::summary_digest(summary),
        });
        start += rows.len();
    }
    let (anchor, proofs) =
        coverage::authenticate(expected.anchor.snapshot_id, events.len(), &commitments);
    if anchor != expected.anchor {
        return Err(invalid_data("rebuilt metadata changed trusted anchor"));
    }
    let sealed =
        coverage::SealedSnapshot::new(events, block_rows, expected.anchor.snapshot_id, None)
            .map_err(|_| invalid_data("trusted builder rejected rebuilt rows"))?;
    if sealed.anchor() != &expected.anchor {
        return Err(invalid_data("rebuilt rows do not match independent anchor"));
    }
    let manifest = Manifest {
        version: VERSION,
        block_rows: block_rows.get(),
        anchor,
        blocks: commitments
            .into_iter()
            .zip(summaries)
            .zip(proofs)
            .map(|((commitment, summary), proof)| BlockMetadata {
                commitment,
                summary,
                proof,
            })
            .collect(),
    };
    write_fresh(&target, &json_bytes(&manifest)?)?;
    Ok(target)
}

pub fn save_checkpoint(
    path: &Path,
    binding: &Binding,
    pages: &[Page],
) -> io::Result<coverage::Digest> {
    let _acc = replay(binding, pages)?;
    let bytes = json_bytes(&CheckpointEnvelope {
        version: VERSION,
        binding: binding.clone(),
        pages: pages.to_vec(),
    })?;
    let digest: coverage::Digest = Sha256::digest(&bytes).into();
    write_fresh(path, &bytes)?;
    Ok(digest)
}

pub fn load_checkpoint(
    path: &Path,
    expected_binding: &Binding,
    expected_digest: coverage::Digest,
) -> io::Result<(Accumulator, Vec<Page>)> {
    let bytes = read_bounded(path)?;
    let digest: coverage::Digest = Sha256::digest(&bytes).into();
    if digest != expected_digest {
        return Err(invalid_data("checkpoint digest mismatch"));
    }
    let envelope: CheckpointEnvelope = serde_json::from_slice(&bytes).map_err(invalid_data)?;
    if envelope.version != VERSION || envelope.binding != *expected_binding {
        return Err(invalid_data("checkpoint version or binding mismatch"));
    }
    let accumulator = replay(expected_binding, &envelope.pages)?;
    Ok((accumulator, envelope.pages))
}

fn replay(binding: &Binding, pages: &[Page]) -> io::Result<Accumulator> {
    let first = pages
        .first()
        .ok_or_else(|| invalid_input("checkpoint history must not be empty"))?;
    if first.binding != *binding {
        return Err(invalid_data("checkpoint page binding mismatch"));
    }
    let mut accumulator = Accumulator::new(binding.clone(), first.clone())
        .map_err(|error| invalid_data(format!("{error:?}")))?;
    for page in &pages[1..] {
        accumulator
            .merge(page.clone())
            .map_err(|error| invalid_data(format!("{error:?}")))?;
    }
    Ok(accumulator)
}

fn excludes(summary: &Summary, query: &Query) -> bool {
    query.start_ns > query.end_ns
        || summary.max_time < query.start_ns
        || summary.min_time > query.end_ns
        || query
            .tenant
            .is_some_and(|tenant| !summary.tenants.contains(&tenant))
        || query
            .token
            .as_ref()
            .is_some_and(|token| !summary.tokens.contains(token))
}

fn matches_query(event: &Event, query: &Query) -> bool {
    let time = event.event_time.0;
    if time < query.start_ns || time > query.end_ns {
        return false;
    }
    if query.tenant.is_some_and(|tenant| tenant != event.tenant.0) {
        return false;
    }
    match (&query.token, &event.payload) {
        (None, _) => true,
        (Some(token), Payload::Log { body }) => body.split_whitespace().any(|word| word == token),
        (Some(_), Payload::Gauge { .. }) => false,
    }
}

struct LockGuard {
    path: PathBuf,
    file: Option<File>,
}
impl LockGuard {
    fn release(mut self, parent: &Path) -> io::Result<()> {
        drop(self.file.take());
        fs::remove_file(&self.path)?;
        sync_dir(parent)
    }
}
impl Drop for LockGuard {
    fn drop(&mut self) {
        let _ = fs::remove_file(&self.path);
    }
}

fn json_bytes<T: Serialize>(value: &T) -> io::Result<Vec<u8>> {
    bounded_json(value)
}

fn bounded_json<T: Serialize>(value: &T) -> io::Result<Vec<u8>> {
    let mut output = BoundedVec(Vec::new());
    serde_json::to_writer(&mut output, value)
        .map_err(|_| invalid_input("encoded file exceeds 64 MiB or cannot be serialized"))?;
    Ok(output.0)
}

struct BoundedVec(Vec<u8>);
impl Write for BoundedVec {
    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
        let Some(new_len) = self.0.len().checked_add(bytes.len()) else {
            return Err(io::Error::new(
                io::ErrorKind::Other,
                "encoded output length overflow",
            ));
        };
        if new_len > MAX_FILE_BYTES {
            return Err(io::Error::new(
                io::ErrorKind::Other,
                "encoded output exceeds limit",
            ));
        }
        self.0.extend_from_slice(bytes);
        Ok(bytes.len())
    }

    fn flush(&mut self) -> io::Result<()> {
        Ok(())
    }
}

fn read_bounded(path: &Path) -> io::Result<Vec<u8>> {
    let file = File::open(path)?;
    let size = file.metadata()?.len();
    if size > MAX_FILE_BYTES as u64 {
        return Err(invalid_input("file exceeds 64 MiB"));
    }
    let mut bytes = Vec::with_capacity(size as usize);
    file.take((MAX_FILE_BYTES + 1) as u64)
        .read_to_end(&mut bytes)?;
    if bytes.len() > MAX_FILE_BYTES {
        return Err(invalid_input("file exceeds 64 MiB"));
    }
    Ok(bytes)
}

fn write_fresh(path: &Path, bytes: &[u8]) -> io::Result<()> {
    if bytes.len() > MAX_FILE_BYTES {
        return Err(invalid_input("file exceeds 64 MiB"));
    }
    let parent = path
        .parent()
        .filter(|parent| !parent.as_os_str().is_empty())
        .unwrap_or(Path::new("."));
    if !parent.is_dir() {
        return Err(invalid_input("file parent must already exist"));
    }
    write_new_synced(path, bytes)?;
    sync_dir(parent)
}

fn write_new_synced(path: &Path, bytes: &[u8]) -> io::Result<()> {
    let mut file = OpenOptions::new().write(true).create_new(true).open(path)?;
    file.write_all(bytes)?;
    file.sync_all()
}

fn sync_dir(path: &Path) -> io::Result<()> {
    File::open(path)?.sync_all()
}

fn invalid_data(error: impl std::fmt::Display) -> io::Error {
    io::Error::new(io::ErrorKind::InvalidData, error.to_string())
}
fn invalid_input(message: &'static str) -> io::Error {
    io::Error::new(io::ErrorKind::InvalidInput, message)
}
fn already_exists(message: &'static str) -> io::Error {
    io::Error::new(io::ErrorKind::AlreadyExists, message)
}
