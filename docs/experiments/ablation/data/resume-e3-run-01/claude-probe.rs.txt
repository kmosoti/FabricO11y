//! Independent adversarial probes for candidate B's E3R resume module,
//! written against tools/storage-probe/RESUME_API.md and
//! docs/experiments/ablation/resume-e3-protocol.md. Not derived from
//! tests/resume_contract.rs or tests/resume_e3.rs (read only for orientation).

use fabric_o11y::{Event, EventId, EventTime, ObservedTime, Payload, ResourceId, SourceId, TenantId};
use std::num::NonZeroUsize;
use storage_probe::Query;
use storage_probe::coverage::{self, CoverageStatus, Disposition, SealedSnapshot};
use storage_probe::resume::{self, Accumulator, Binding, MatchedRow, Residual, ResumeError};

fn event(id: u64, time: i64, body: &str) -> Event {
    Event {
        id: EventId(id),
        tenant: TenantId(1),
        source: SourceId(2),
        resource: ResourceId(3),
        event_time: EventTime(time),
        observed_time: ObservedTime(time),
        attributes: vec![],
        payload: Payload::Log { body: body.into() },
    }
}

fn q_all() -> Query {
    Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: None }
}

fn seal(rows: Vec<Event>, id: u64, size: usize) -> SealedSnapshot {
    SealedSnapshot::new(rows, NonZeroUsize::new(size).unwrap(), id, None).unwrap()
}

fn binding(s: &SealedSnapshot, query: Query) -> Binding {
    Binding {
        anchor: s.anchor().clone(),
        query,
        tokenizer_version: resume::TOKENIZER_VERSION,
        order_version: resume::ORDER_VERSION,
    }
}

fn assert_unchanged(acc: &Accumulator, rows: &[MatchedRow], status: &CoverageStatus, residual: &Residual) {
    assert_eq!(acc.rows(), rows);
    assert_eq!(&acc.status(), status);
    assert_eq!(&acc.residual(), residual);
}

fn snapshot_state(acc: &Accumulator) -> (Vec<MatchedRow>, CoverageStatus, Residual) {
    (acc.rows(), acc.status(), acc.residual())
}

// === 1) Full snapshot/query/version binding =================================

#[test]
fn binding_mismatches_on_every_axis_are_rejected_before_execution() {
    let rows = vec![event(1, 0, "a"), event(2, 1, "b"), event(3, 2, "c"), event(4, 3, "d")];
    let other_rows = vec![event(1, 0, "a"), event(2, 1, "b"), event(3, 2, "c"), event(4, 3, "d")];
    let snap = seal(rows, 501, 2);
    let other_snap = seal(other_rows, 502, 2);
    let b = binding(&snap, q_all());

    // Wrong tokenizer/order version -> WrongBinding, caught before any query runs.
    let mut wrong_tok = b.clone();
    wrong_tok.tokenizer_version += 1;
    assert_eq!(resume::execute(&snap, &wrong_tok, &[true, true]), Err(ResumeError::WrongBinding));
    let mut wrong_ord = b.clone();
    wrong_ord.order_version += 1;
    assert_eq!(resume::execute(&snap, &wrong_ord, &[true, true]), Err(ResumeError::WrongBinding));

    // Wrong snapshot (different anchor entirely) -> WrongBinding.
    assert_eq!(
        resume::execute(&other_snap, &b, &[true, true]),
        Err(ResumeError::WrongBinding)
    );

    // Anchor field tampering (snapshot_id, root, row_count, block_count) all reject.
    for mutate in [
        (|a: &mut storage_probe::coverage::Anchor| a.snapshot_id += 1) as fn(&mut storage_probe::coverage::Anchor),
        |a| a.root[0] ^= 1,
        |a| a.row_count += 1,
        |a| a.block_count += 1,
    ] {
        let mut tampered = b.clone();
        mutate(&mut tampered.anchor);
        assert_eq!(resume::execute(&snap, &tampered, &[true, true]), Err(ResumeError::WrongBinding));
    }

    // Availability length mismatch is a Coverage(InvalidLayout) error, not silently
    // truncated/padded.
    assert!(resume::execute(&snap, &b, &[true]).is_err());
    assert!(resume::execute(&snap, &b, &[true, true, true]).is_err());
}

// === 2) Residual subsets =====================================================

#[test]
fn residual_must_be_sorted_unique_and_match_authenticated_commitments() {
    let rows = vec![event(1, 0, "a"), event(2, 1, "b"), event(3, 2, "c"), event(4, 3, "d")];
    let snap = seal(rows, 601, 1);
    let b = binding(&snap, q_all());
    let page = resume::execute(&snap, &b, &[false; 4]).unwrap();
    let acc = Accumulator::new(b.clone(), page).unwrap();
    let full_residual = acc.residual();
    assert_eq!(full_residual.blocks.len(), 4);

    // Unsorted (descending pair) rejected.
    let mut unsorted = full_residual.clone();
    unsorted.blocks.swap(0, 1);
    assert_eq!(resume::resume(&snap, &unsorted, &[true; 4]), Err(ResumeError::InvalidResidual));

    // Duplicate ordinal (same commitment twice) rejected -- ordinal <= prev catches equals.
    let duplicated = Residual { binding: b.clone(), blocks: vec![full_residual.blocks[0].clone(), full_residual.blocks[0].clone()] };
    assert_eq!(resume::resume(&snap, &duplicated, &[true; 4]), Err(ResumeError::InvalidResidual));

    // Out-of-range ordinal rejected.
    let mut bad_ordinal = full_residual.blocks[0].clone();
    bad_ordinal.ordinal = 4;
    let oob = Residual { binding: b.clone(), blocks: vec![bad_ordinal] };
    assert_eq!(resume::resume(&snap, &oob, &[true; 4]), Err(ResumeError::InvalidResidual));

    // Genuine (not authenticated) commitment content for a real ordinal rejected.
    let mut forged = full_residual.blocks[0].clone();
    forged.rows_digest[0] ^= 1;
    let forged_residual = Residual { binding: b.clone(), blocks: vec![forged] };
    assert_eq!(resume::resume(&snap, &forged_residual, &[true; 4]), Err(ResumeError::InvalidResidual));

    // A genuinely sorted, unique, authenticated subset succeeds and scans only
    // the requested blocks that are ALSO available; requested-but-unavailable
    // stays Unavailable in the returned page, never silently promoted.
    let subset = Residual {
        binding: b.clone(),
        blocks: vec![full_residual.blocks[0].clone(), full_residual.blocks[2].clone()],
    };
    // Only block 0 available; block 2 requested but not available.
    let page = resume::resume(&snap, &subset, &[true, false, false, false]).unwrap();
    assert!(matches!(page.receipt.blocks[0].disposition, Disposition::Scanned));
    assert!(matches!(page.receipt.blocks[2].disposition, Disposition::Unavailable));
    // Block 1 and 3 were never requested; permitted stays false regardless of
    // availability, so they must also be Unavailable in this page.
    assert!(matches!(page.receipt.blocks[1].disposition, Disposition::Unavailable));
    assert!(matches!(page.receipt.blocks[3].disposition, Disposition::Unavailable));
}

// === 3) Conflicting entire-block results ====================================

#[test]
fn conflicting_entire_block_result_is_rejected_and_state_is_unchanged() {
    let rows = vec![event(1, 0, "x"), event(2, 1, "x")];
    let query = Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("x".into()) };
    let snap = seal(rows, 701, 2);
    let b = binding(&snap, query);
    let first = resume::execute(&snap, &b, &[true]).unwrap();
    let mut acc = Accumulator::new(b.clone(), first.clone()).unwrap();
    let (rows0, status0, residual0) = snapshot_state(&acc);

    // A same-block "resolution" that drops one genuinely matching row entirely
    // (an entire-block conflict, not a single-field mutation) must be rejected,
    // and the accumulator must be byte-for-byte unchanged afterward.
    let mut shrunk = first.clone();
    shrunk.rows.truncate(1);
    let result = acc.merge(shrunk);
    assert_eq!(result, Err(ResumeError::Conflict));
    assert_unchanged(&acc, &rows0, &status0, &residual0);

    // A same-block resolution reporting a totally different (but structurally
    // valid) row set for the same Scanned block also conflicts.
    let mut swapped_digests = first;
    swapped_digests.rows[0].digest[31] ^= 0xFF;
    swapped_digests.rows[1].digest[31] ^= 0xFF;
    let result2 = acc.merge(swapped_digests);
    assert_eq!(result2, Err(ResumeError::Conflict));
    assert_unchanged(&acc, &rows0, &status0, &residual0);
}

// === 4) Atomic rejection even after new earlier blocks ======================

#[test]
fn merge_does_not_partially_apply_a_new_earlier_block_when_a_later_block_conflicts() {
    let rows = vec![
        event(1, 0, "m"), // block 0
        event(2, 1, "m"), // block 1 (new, will be introduced in the poison page)
        event(3, 2, "m"), // block 2 (already resolved, will conflict)
    ];
    let query = Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("m".into()) };
    let snap = seal(rows, 801, 1);
    let b = binding(&snap, query);

    // Step 1: resolve only block 0.
    let page_a = resume::execute(&snap, &b, &[true, false, false]).unwrap();
    let mut acc = Accumulator::new(b.clone(), page_a).unwrap();

    // Step 2: resolve block 2 as well (block 1 stays unresolved).
    let page_b = resume::execute(&snap, &b, &[true, false, true]).unwrap();
    acc.merge(page_b).unwrap();
    assert_eq!(acc.status(), CoverageStatus::Incomplete { unavailable: vec![1] });
    let (rows_before, status_before, residual_before) = snapshot_state(&acc);
    assert!(residual_before.blocks.iter().any(|c| c.ordinal == 1));

    // Step 3 (poison page): block 0 unchanged (consistent), block 1 is a BRAND
    // NEW resolution (previously unresolved, "earlier" than the conflict at
    // ordinal 2), and block 2 is corrupted to conflict with its already-
    // resolved value. The whole merge must fail, and critically block 1 must
    // NOT have been silently absorbed just because it came before the conflict.
    let mut poison = resume::execute(&snap, &b, &[true, true, true]).unwrap();
    assert!(
        poison.receipt.blocks[2]
            .disposition
            .eq(&Disposition::Scanned)
    );
    // Corrupt block 2's row digest so it conflicts with the already-resolved value.
    let block2_row_index = poison.rows.iter().position(|r| r.block == 2).expect("block 2 has a row");
    poison.rows[block2_row_index].digest[0] ^= 1;

    let result = acc.merge(poison);
    assert_eq!(result, Err(ResumeError::Conflict));

    // The accumulator must be completely unchanged: block 1 must still be
    // unresolved even though its (valid, non-conflicting) resolution was
    // present in the same rejected page.
    assert_unchanged(&acc, &rows_before, &status_before, &residual_before);
    assert_eq!(acc.status(), CoverageStatus::Incomplete { unavailable: vec![1] });
}

// === 5) Duplicate-identical rows, distinct positions =========================

#[test]
fn identical_bodies_at_distinct_positions_are_never_collapsed() {
    let rows = vec![event(9, 0, "dup"), event(9, 0, "dup"), event(9, 0, "dup")];
    let query = Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("dup".into()) };
    let snap = seal(rows, 901, 3);
    let b = binding(&snap, query);
    let page = resume::execute(&snap, &b, &[true]).unwrap();
    let acc = Accumulator::new(b, page).unwrap();
    assert_eq!(acc.positions(), vec![0, 1, 2]);
    let all_rows = acc.rows();
    assert_eq!(all_rows.len(), 3);
    // All three rows share the same EventId and same row digest (identical
    // content) but must remain three distinct (block, offset) coordinates.
    let digests: Vec<_> = all_rows.iter().map(|r| r.digest).collect();
    assert_eq!(digests[0], digests[1]);
    assert_eq!(digests[1], digests[2]);
    let offsets: Vec<_> = all_rows.iter().map(|r| r.offset).collect();
    assert_eq!(offsets, vec![0, 1, 2]);
}

// === 6) Replay / reorder / idempotence =======================================

#[test]
fn identical_retry_is_idempotent_and_merge_order_is_immaterial() {
    let rows = vec![event(1, 0, "z"), event(2, 1, "z"), event(3, 2, "z"), event(4, 3, "z")];
    let query = Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("z".into()) };
    let snap = seal(rows, 1001, 1);
    let b = binding(&snap, query);
    let block0 = resume::execute(&snap, &b, &[true, false, false, false]).unwrap();
    let block1 = resume::execute(&snap, &b, &[false, true, false, false]).unwrap();

    // Order A: block0 then block1, then replay block0 (identical retry).
    let mut acc_a = Accumulator::new(b.clone(), block0.clone()).unwrap();
    acc_a.merge(block1.clone()).unwrap();
    let before_replay = snapshot_state(&acc_a);
    acc_a.merge(block0.clone()).unwrap(); // identical retry
    assert_eq!(snapshot_state(&acc_a), before_replay, "identical retry must be a no-op");

    // Order B: block1 first, then block0 -- must converge to the same state.
    let mut acc_b = Accumulator::new(b.clone(), block1).unwrap();
    acc_b.merge(block0).unwrap();

    assert_eq!(acc_a.rows(), acc_b.rows());
    assert_eq!(acc_a.status(), acc_b.status());
    assert_eq!(acc_a.residual(), acc_b.residual());
    assert_eq!(acc_a.positions(), acc_b.positions());
}

// === 7) Unresolved vs empty resolved; later Unavailable cannot undo =========

#[test]
fn resolved_empty_block_cannot_be_reverted_to_unresolved_by_a_later_page() {
    // Use a time window that overlaps the block's coarse summary range
    // (min=0, max=10) so the block is never metadata-excluded, but matches
    // zero actual rows (neither event's exact time -- 0 or 10 -- falls in
    // [3,7]). This isolates a genuinely availability-dependent block: the
    // SAME query is Unavailable when unavailable and a RESOLVED (Scanned,
    // empty) result when available, unlike a summary-excluded block whose
    // disposition would be identical either way.
    let rows = vec![event(1, 0, "x"), event(2, 10, "x")];
    let query = Query { start_ns: 3, end_ns: 7, tenant: None, token: None };
    let snap = seal(rows, 1101, 2);
    let b = binding(&snap, query);

    // Block 0 (only block) is unavailable in the first page -> genuinely unresolved.
    let unresolved_page = resume::execute(&snap, &b, &[false]).unwrap();
    assert!(matches!(unresolved_page.receipt.blocks[0].disposition, Disposition::Unavailable));
    let mut acc = Accumulator::new(b.clone(), unresolved_page).unwrap();
    assert_eq!(acc.status(), CoverageStatus::Incomplete { unavailable: vec![0] });
    assert!(acc.rows().is_empty());

    // Now resolve it: available, but zero rows actually match -> Scanned, empty.
    // This is a RESOLVED empty result, distinct from the unresolved state above.
    let resolved_page = resume::execute(&snap, &b, &[true]).unwrap();
    assert!(matches!(resolved_page.receipt.blocks[0].disposition, Disposition::Scanned));
    assert!(resolved_page.rows.is_empty());
    acc.merge(resolved_page).unwrap();
    assert_eq!(acc.status(), CoverageStatus::Complete);
    assert!(acc.residual().blocks.is_empty());

    // A later page claiming this same (already-resolved) block is now
    // Unavailable must NOT undo the resolution -- merge must succeed (it is
    // not a conflict, since Unavailable maps to "no new information") and the
    // block must remain resolved/Complete afterward.
    let regressive_page = resume::execute(&snap, &b, &[false]).unwrap();
    assert!(matches!(regressive_page.receipt.blocks[0].disposition, Disposition::Unavailable));
    acc.merge(regressive_page).expect("a later Unavailable claim must not error");
    assert_eq!(
        acc.status(),
        CoverageStatus::Complete,
        "a later Unavailable page must not revert an already-resolved block to unresolved"
    );
    assert!(acc.residual().blocks.is_empty());
}

// === 8) Trusted-executor limits (documented, not a bug) =====================

#[test]
fn accumulator_cannot_itself_detect_a_fabricated_but_structurally_valid_row() {
    // RESUME_API.md: "The executor is trusted to report the entire exact match
    // list for each Scanned block; metadata verification does not prove
    // execution." Accumulator::new/merge take no &SealedSnapshot and so cannot
    // re-derive row_digest_at() themselves. Confirm this documented limit is
    // real: a Page reporting a structurally well-formed but fabricated
    // MatchedRow (wrong digest, or a position that doesn't actually match the
    // query) for a genuinely Scanned block is accepted without error by
    // Accumulator::new. This is expected scope, not something we flag as a
    // defect -- only execute()/resume() (which DO have snapshot access) bind
    // real digests via row_digest_at.
    let rows = vec![event(1, 0, "real")];
    let query = Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("real".into()) };
    let snap = seal(rows, 1201, 1);
    let b = binding(&snap, query);
    let mut page = resume::execute(&snap, &b, &[true]).unwrap();
    assert_eq!(page.rows.len(), 1);
    // Fabricate the digest to something that does not correspond to any real
    // row content (Accumulator has no way to check this).
    page.rows[0].digest = [0xAB; 32];
    let acc = Accumulator::new(b, page);
    assert!(
        acc.is_ok(),
        "Accumulator::new has no snapshot access and cannot itself verify row \
         content against the real snapshot -- this is the documented trust \
         boundary, not something this layer is expected to catch"
    );
}

#[test]
fn coverage_verify_alone_cannot_prove_a_scanned_block_was_actually_scanned() {
    // Same limit restated at the coverage layer: verify() only checks
    // metadata/Merkle authentication, never row content, for Disposition::Scanned.
    let rows = vec![event(1, 0, "present")];
    let query = Query { start_ns: i64::MIN, end_ns: i64::MAX, tenant: None, token: Some("present".into()) };
    let snap = seal(rows, 1301, 1);
    let answer = snap.query(&query, &[true]).unwrap();
    // The receipt alone (no row content) still verifies fine, independent of
    // whether the executor actually matched anything correctly.
    assert_eq!(
        coverage::verify(snap.anchor(), &query, &answer.receipt),
        Ok(CoverageStatus::Complete)
    );
}
